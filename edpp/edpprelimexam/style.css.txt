/* ===== ANIMATED GRADIENT BACKGROUND ===== */
/* This creates the colorful, moving background effect */

.gradient-background {
    /* Styles the outer container div with the animated gradient */
    
    position: relative;
    /* Sets positioning context for absolutely positioned children */
    /* The .mainbg div uses this as its reference point */
    
    width: 100%;
    /* Takes up the full width of the viewport */
    
    height: 100vh;
    /* Takes up 100% of viewport height (full screen) */
    /* vh stands for "viewport height" */
    
    background: linear-gradient(45deg, #ff9a9e, #fecfef, #fefeff, #c3eefb, #ffc3a0);
    /* Creates a diagonal gradient with 5 color stops */
    /* 45deg makes it go from bottom-left to top-right */
    
    background-size: 400% 400%;
    /* Makes the background 4x larger than the container */
    /* This allows the background to move during animation */
    
    display: flex;
    /* Enables flexbox layout for centering */
    
    justify-content: center;
    /* Centers child elements horizontally */
    
    align-items: center;
    /* Centers child elements vertically */
    
    animation: gradientShift 15s ease infinite;
    /* Applies the gradientShift animation */
    /* 15s = duration, ease = timing function, infinite = repeats forever */
    /* This makes the gradient colors move smoothly */
}

@keyframes gradientShift {
    /* Defines the animation frames for the moving gradient */
    
    0% {
        /* At the start of the animation */
        background-position: 0% 50%;
        /* Positions the gradient at the left side */
    }
    
    50% {
        /* Halfway through the animation */
        background-position: 100% 50%;
        /* Moves the gradient to the right side */
    }
    
    100% {
        /* At the end of the animation */
        background-position: 0% 50%;
        /* Returns to the starting position */
        /* This creates a seamless loop */
    }
}
